FUNCTIONALITY:
Import supabase from '../supabaseClient.js'

On "Send OTP" press:
1. Validate phone number (must be 10 digits)
2. Format as: +880XXXXXXXXXX
3. Call: await supabase.auth.signInWithOtp({ phone: formattedPhone })
4. If success: Show OTP input section
5. If error: Show error message

On "Verify & Login" press:
1. Validate OTP (must be 6 digits)
2. Call: await supabase.auth.verifyOtp({ phone: formattedPhone, token: otpCode, type: 'sms' })
3. If success: User is logged in (Supabase handles session automatically)
4. If error: Show error message

2. Modify App.js:

Import AuthScreen and supabase
Add authentication state management using useEffect and useState
Check if user is logged in on app load
If logged in: Show text "স্বাগতম! লগইন সফল হয়েছে" (Welcome! Login successful)
If not logged in: Show AuthScreen

Use this pattern:
```javascript
import { useEffect, useState } from 'react';
import { supabase } from './supabaseClient';
import AuthScreen from './screens/AuthScreen';

export default function App() {
  const [session, setSession] = useState(null);

  useEffect(() => {
    supabase.auth.getSession().then(({ data: { session } }) => {
      setSession(session);
    });

    const { data: { subscription } } = supabase.auth.onAuthStateChange((_event, session) => {
      setSession(session);
    });

    return () => subscription.unsubscribe();
  }, []);

  if (!session) {
    return <AuthScreen />;
  }

  return (
    <View style={{ flex: 1, justifyContent: 'center', alignItems: 'center', backgroundColor: '#F9F9F9' }}>
      <Text style={{ fontSize: 24, fontWeight: 'bold' }}>স্বাগতম! লগইন সফল হয়েছে</Text>
    </View>
  );
}
```
